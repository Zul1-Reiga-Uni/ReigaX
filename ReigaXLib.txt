-- theme can be a preset name, or a table (optional Base = preset name, then overrides)
local function resolveTheme(t)
    local out = {}
    for k, v in pairs(ReigaX.Themes.Default) do out[k] = v end

    local function apply(src)
        for k, v in pairs(src) do
            if k ~= "Base" then out[k] = v end
        end
    end

    if type(t) == "string" then
        if ReigaX.Themes[t] then apply(ReigaX.Themes[t]) end
    elseif type(t) == "table" then
        if type(t.Base) == "string" and ReigaX.Themes[t.Base] then
            apply(ReigaX.Themes[t.Base])
        end
        apply(t)
    end
    return out
end

-- helpers
local function new(class, props, children)
    local o = Instance.new(class)
    for k, v in pairs(props or {}) do o[k] = v end
    for _, c in ipairs(children or {}) do c.Parent = o end
    return o
end

local function tween(o, t, p, style, dir)
    local tw = TweenService:Create(o, TweenInfo.new(t, style or EASE.Quad, dir or DIR.Out), p)
    tw:Play()
    return tw
end

-- run callback without breaking the ui
local function safe(fn, ...)
    local ok, err = pcall(fn, ...)
    if not ok then warn("[ReigaX] callback error: " .. tostring(err)) end
end

local function getGuiParent()
    local ok, h = pcall(function() return gethui and gethui() end)
    if ok and h then return h end
    local ok2, cg = pcall(function()
        local c = game:GetService("CoreGui")
        local t = Instance.new("Folder"); t.Parent = c; t:Destroy()
        return c
    end)
    if ok2 and cg then return cg end
    return LP:WaitForChild("PlayerGui")
end

-- 1 -> "1", 1.5 -> "1.5", 1.234 -> "1.23"
local function fmtNumber(v)
    if v == math.floor(v) then return tostring(math.floor(v)) end
    local s = string.format("%.2f", v)
    s = s:gsub("0+$", ""):gsub("%.$", "")
    return s
end

-- section open / close (height tween, things below follow)
local function animateGroup(group, layout, open)
    local token = (group:GetAttribute("Tok") or 0) + 1
    group:SetAttribute("Tok", token)
    local info = TweenInfo.new(0.32, EASE.Quint, DIR.Out)

    if open then
        local startH = group.Visible and group.AbsoluteSize.Y or 0
        group.AutomaticSize = Enum.AutomaticSize.None
        group.ClipsDescendants = true
        group.Size = UDim2.new(1, 0, 0, startH)
        group.Visible = true
        local target = layout.AbsoluteContentSize.Y
        if target <= 0 then
            -- layout not ready yet, wait a frame
            RunService.Heartbeat:Wait()
            if group:GetAttribute("Tok") ~= token then return end
            target = layout.AbsoluteContentSize.Y
        end
        local tw = TweenService:Create(group, info, { Size = UDim2.new(1, 0, 0, target) })
        tw.Completed:Connect(function()
            if group:GetAttribute("Tok") == token then
                group.ClipsDescendants = false
                group.AutomaticSize = Enum.AutomaticSize.Y
                group.Size = UDim2.new(1, 0, 0, 0)
            end
        end)
        tw:Play()
    else
        local cur = group.AbsoluteSize.Y
        group.AutomaticSize = Enum.AutomaticSize.None
        group.ClipsDescendants = true
        group.Size = UDim2.new(1, 0, 0, cur)
        local tw = TweenService:Create(group, info, { Size = UDim2.new(1, 0, 0, 0) })
        tw.Completed:Connect(function()
            if group:GetAttribute("Tok") == token then
                group.Visible = false
                group.ClipsDescendants = false
                group.AutomaticSize = Enum.AutomaticSize.Y
                group.Size = UDim2.new(1, 0, 0, 0)
            end
        end)
        tw:Play()
    end
end

    if activeNotify then activeNotify(cfg or {}) end
end

-- kick if the game is not allowed. id can be a number or a table of numbers
function ReigaX:CheckGame(ids, message)
    if type(ids) ~= "table" then ids = { ids } end
    for _, id in ipairs(ids) do
        if game.GameId == id then return true end
    end
    LP:Kick(message or "Wrong Game!")
    return false
end

        return new("UICorner", { CornerRadius = UDim.new(0, r or 8) })

        return new("UIPadding", {
            PaddingLeft = UDim.new(0, l or 0), PaddingRight = UDim.new(0, r or 0),
            PaddingTop = UDim.new(0, t or 0), PaddingBottom = UDim.new(0, b or 0),

    local function gradient(rot)
        return new("UIGradient", {
            Color = ColorSequence.new(T.Primary, T.Secondary),

        return new("UIStroke", {
            Thickness = thickness or 1.5, Color = WHITE, Transparency = transparency or 0.35,

    -- rotating gradient border
    local function spin(stroke, duration)
        if cfg.AnimatedBorder == false then return end
        local g = stroke:FindFirstChildOfClass("UIGradient")
        if not g then return end
        local tw = TweenService:Create(
            g,
            TweenInfo.new(duration or 5, EASE.Linear, DIR.Out, -1),
            { Rotation = g.Rotation + 360 }
        )
        tw:Play()
        track(tw)
    end

    -- press effect (text shrinks then bounces back)
    local function pressFx(btn, label)
        label = label or btn
        local base = label.TextSize
        btn.MouseButton1Down:Connect(function()
            tween(label, 0.08, { TextSize = base - 2 })
        end)
        local function release()
            tween(label, 0.28, { TextSize = base }, EASE.Back, DIR.Out)
        end
        btn.MouseButton1Up:Connect(release)
        btn.MouseLeave:Connect(release)
    end

    -- hover + press scale (anchor point must be 0.5, 0.5)
    local function scaleFx(btn, hoverScale, downScale)
        local sc = new("UIScale", { Scale = 1, Parent = btn })
        btn.MouseEnter:Connect(function() tween(sc, 0.15, { Scale = hoverScale }) end)
        btn.MouseLeave:Connect(function() tween(sc, 0.22, { Scale = 1 }) end)
        btn.MouseButton1Down:Connect(function() tween(sc, 0.08, { Scale = downScale }) end)
        btn.MouseButton1Up:Connect(function()
            tween(sc, 0.3, { Scale = hoverScale }, EASE.Back, DIR.Out)
        end)
        return sc
    end

    -- smooth drag. returns a function that tells if the last press was a drag
    local function makeDraggable(handle, target, threshold)
        threshold = threshold or 0
        local dragging, moved, startInput, startPos, goal = false, false, nil, nil, nil

        handle.InputBegan:Connect(function(i)
            if i.UserInputType == Enum.UserInputType.MouseButton1
            or i.UserInputType == Enum.UserInputType.Touch then
                dragging   = true
                moved      = false
                startInput = i.Position
                startPos   = target.Position
                goal       = nil
                i.Changed:Connect(function()
                    if i.UserInputState == Enum.UserInputState.End then dragging = false end
                end)
            end
        end)

        track(UserInputService.InputChanged:Connect(function(i)
            if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement
            or i.UserInputType == Enum.UserInputType.Touch) then
                local d = i.Position - startInput
                if d.Magnitude > threshold then moved = true end
                if moved then
                    goal = UDim2.new(
                        startPos.X.Scale, startPos.X.Offset + d.X,
                        startPos.Y.Scale, startPos.Y.Offset + d.Y
                    )
                end
            end
        end))

        track(RunService.RenderStepped:Connect(function(dt)
            if not goal then return end
            local nxt = target.Position:Lerp(goal, 1 - math.exp(-22 * dt))
            if math.abs(nxt.X.Offset - goal.X.Offset) < 1.5
            and math.abs(nxt.Y.Offset - goal.Y.Offset) < 1.5 then
                nxt = goal
                if not dragging then goal = nil end
            end
            target.Position = nxt
        end))

        return function() return moved end
    end

    --// config saving

    local function csInfo()
        local cs = cfg.ConfigurationSaving
        if not (cs and cs.Enabled) then return nil end
        local folder = cs.FolderName or "ReigaX"
        local file = (cs.FileName or tostring(game.GameId)) .. ".json"
        return folder, folder .. "/" .. file
    end

    local function changed()
        if not csInfo() then return end
        saveToken = saveToken + 1
        local mine = saveToken
        task.delay(0.6, function()
            if mine == saveToken and not destroyed then Window:SaveConfiguration() end
        end)
    end

    local function register(flag, element)
        if not flag then return end
        flagList[flag] = element
        ReigaX.Flags[flag] = element
    end

    function Window:SaveConfiguration()
        local folder, path = csInfo()
        if not path or not writefile then return false end
        pcall(function()
            if isfolder and makefolder and not isfolder(folder) then makefolder(folder) end
        end)
        local data = {}
        for flag, element in pairs(flagList) do
            local encode = element._encode
            local ok, v = pcall(function()
                if encode then return encode(element.CurrentValue) end
                return element.CurrentValue
            end)
            if ok then data[flag] = v end
        end
        local ok = pcall(function() writefile(path, HttpService:JSONEncode(data)) end)
        return ok
    end

    function Window:LoadConfiguration()
        local folder, path = csInfo()
        if not path or not readfile or not isfile then return false end
        local okFile, exists = pcall(isfile, path)
        if not okFile or not exists then return false end
        local ok, data = pcall(function() return HttpService:JSONDecode(readfile(path)) end)
        if not ok or type(data) ~= "table" then return false end
        for flag, raw in pairs(data) do
            local element = flagList[flag]
            if element then
                pcall(function()
                    local value = raw
                    if element._decode then value = element._decode(raw) end
                    element:Set(value)
                end)
            end
        end
        return true
    end

        tween(p, 0.2, { GroupTransparency = 1, Position = out }, EASE.Quad, DIR.In)
        task.delay(0.22, function() pcall(function() p:Destroy() end) end)
    end

        function obj:CreateInput(o)
            local r = row(30)
            rowLabel(r, o.Name, 100)

            local boxStroke = new("UIStroke", { Color = T.Primary, Thickness = 1, Transparency = 0.6 })
            local startValue
            if o.NumbersOnly then
                startValue = tonumber(o.CurrentValue) or 0
            else
                startValue = tostring(o.CurrentValue or "")
            end
            local box = new("TextBox", {
                Size = UDim2.fromOffset(90, 22), Position = UDim2.new(1, -100, 0.5, -11),
                BackgroundColor3 = T.Bg, Text = tostring(startValue),
                PlaceholderText = tostring(o.PlaceholderText or ""), PlaceholderColor3 = T.Sub,
                TextColor3 = T.Primary, Font = Enum.Font.GothamMedium, TextSize = 12,
                ClearTextOnFocus = false, Parent = r,
            }, { corner(6), boxStroke })

            local Input = { Type = "Input", CurrentValue = startValue }

            local function apply(text, silent)
                local value = text
                if o.NumbersOnly then
                    local n = tonumber(text)
                    if n == nil then
                        box.Text = tostring(Input.CurrentValue)
                        return
                    end
                    if o.Min then n = math.max(o.Min, n) end
                    if o.Max then n = math.min(o.Max, n) end
                    value = n
                    box.Text = tostring(n)
                end
                Input.CurrentValue = value
                if not silent and o.Callback then task.spawn(safe, o.Callback, value) end
                changed()
                if o.RemoveTextAfterFocusLost then box.Text = "" end
            end

            box.Focused:Connect(function()
                tween(boxStroke, 0.2, { Transparency = 0, Thickness = 1.8 })
            end)
            box.FocusLost:Connect(function()
                tween(boxStroke, 0.3, { Transparency = 0.6, Thickness = 1 })
                apply(box.Text)
            end)

            function Input:Set(v, silent)
                box.Text = tostring(v)
                apply(box.Text, silent)
            end
            function Input:Get() return Input.CurrentValue end

            register(o.Flag, Input)
            return Input
        end

        --// slider
        function obj:CreateSlider(o)
            local range = o.Range or { 0, 100 }
            local minV, maxV = range[1], range[2]
            local inc = o.Increment or 1
            local suffix = o.Suffix and (" " .. tostring(o.Suffix)) or ""

            local r = row(40)
            rowLabel(r, o.Name, 90).Size = UDim2.new(1, -100, 0, 22)
            local valLbl = new("TextLabel", {
                Size = UDim2.fromOffset(80, 22), Position = UDim2.new(1, -90, 0, 0),
                BackgroundTransparency = 1, Text = "", TextColor3 = T.Primary,
                Font = Enum.Font.GothamMedium, TextSize = 11,
                TextXAlignment = Enum.TextXAlignment.Right, Parent = r,
            })
            local bar = new("Frame", {
                Size = UDim2.new(1, -20, 0, 6), Position = UDim2.new(0, 10, 0, 28),
                BackgroundColor3 = T.Off, BorderSizePixel = 0, Parent = r,
            }, { corner(3) })
            local fill = new("Frame", {
                Size = UDim2.fromScale(0, 1), BackgroundColor3 = WHITE,
                BorderSizePixel = 0, Parent = bar,
            }, { corner(3), gradient() })
            local knob = new("Frame", {
                Size = UDim2.fromOffset(12, 12), AnchorPoint = Vector2.new(0.5, 0.5),
                Position = UDim2.new(0, 0, 0.5, 0), BackgroundColor3 = WHITE,
                BorderSizePixel = 0, ZIndex = 2, Parent = bar,
            }, { corner(6) })
            local hit = new("TextButton", {
                Size = UDim2.new(1, -10, 0, 22), Position = UDim2.new(0, 5, 0, 20),
                BackgroundTransparency = 1, Text = "", Parent = r,
            })

            local function snap(v)
                v = math.clamp(v, minV, maxV)
                if inc > 0 then
                    v = minV + math.floor((v - minV) / inc + 0.5) * inc
                end
                v = math.clamp(v, minV, maxV)
                return tonumber(string.format("%.6f", v))
            end

            local Slider = { Type = "Slider", CurrentValue = snap(o.CurrentValue or minV) }

            local function render(animate)
                local alpha = 0
                if maxV ~= minV then alpha = (Slider.CurrentValue - minV) / (maxV - minV) end
                valLbl.Text = fmtNumber(Slider.CurrentValue) .. suffix
                if animate then
                    tween(fill, 0.08, { Size = UDim2.fromScale(alpha, 1) })
                    tween(knob, 0.08, { Position = UDim2.new(alpha, 0, 0.5, 0) })
                else
                    fill.Size = UDim2.fromScale(alpha, 1)
                    knob.Position = UDim2.new(alpha, 0, 0.5, 0)
                end
            end

            local function apply(v, silent, onlyIfChanged)
                v = snap(v)
                if onlyIfChanged and v == Slider.CurrentValue then return end
                Slider.CurrentValue = v
                render(true)
                if not silent and o.Callback then task.spawn(safe, o.Callback, v) end
                changed()
            end

            render(false)

            local dragging = false
            local function fromInput(pos)
                local w = bar.AbsoluteSize.X
                if w <= 0 then return end
                local a = math.clamp((pos.X - bar.AbsolutePosition.X) / w, 0, 1)
                apply(minV + (maxV - minV) * a, false, true)
            end

            hit.InputBegan:Connect(function(i)
                if i.UserInputType == Enum.UserInputType.MouseButton1
                or i.UserInputType == Enum.UserInputType.Touch then
                    dragging = true
                    fromInput(i.Position)
                    i.Changed:Connect(function()
                        if i.UserInputState == Enum.UserInputState.End then dragging = false end
                    end)
                end
            end)
            track(UserInputService.InputChanged:Connect(function(i)
                if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement
                or i.UserInputType == Enum.UserInputType.Touch) then
                    fromInput(i.Position)
                end
            end))

            function Slider:Set(v, silent) apply(tonumber(v) or minV, silent, false) end
            function Slider:Get() return Slider.CurrentValue end

            register(o.Flag, Slider)
            return Slider
        end

        --// dropdown, the list opens in a side panel. single -> string, multi -> table
        function obj:CreateDropdown(o)
            local options = o.Options or {}
            local multi = o.MultipleOptions and true or false

            local init = o.CurrentOption
            if type(init) == "string" then init = { init } end
            init = init or {}

            local current = nil          -- single mode
            local chosen = {}            -- multi mode (set)
            if multi then
                for _, v in ipairs(init) do chosen[v] = true end
            else
                current = init[1]
            end

            local r = row(30)
            rowLabel(r, o.Name, 150)

            local sStroke = new("UIStroke", { Color = T.Primary, Thickness = 1, Transparency = 0.5 })
            local btn = new("TextButton", {
                Size = UDim2.fromOffset(140, 22), Position = UDim2.new(1, -150, 0.5, -11),
                BackgroundColor3 = T.Bg, Text = "", AutoButtonColor = false, Parent = r,
            }, { corner(6), sStroke })
            local val = new("TextLabel", {
                Size = UDim2.new(1, -24, 1, 0), Position = UDim2.new(0, 8, 0, 0),
                BackgroundTransparency = 1, Text = "", TextColor3 = T.Primary,
                Font = Enum.Font.GothamMedium, TextSize = 11,
                TextXAlignment = Enum.TextXAlignment.Left,
                TextTruncate = Enum.TextTruncate.AtEnd, Parent = btn,
            })
            local arrow = new("TextLabel", {
                Size = UDim2.fromOffset(16, 22), Position = UDim2.new(1, -18, 0, 0),
                BackgroundTransparency = 1, Text = ">", TextColor3 = T.Sub,
                Font = Enum.Font.GothamBold, TextSize = 12, Parent = btn,
            })

            pressFx(btn, val)
            btn.MouseEnter:Connect(function() tween(sStroke, 0.15, { Transparency = 0 }) end)
            btn.MouseLeave:Connect(function() tween(sStroke, 0.2, { Transparency = 0.5 }) end)

            local Dropdown = { Type = "Dropdown" }

            local function isOn(name)
                if multi then return chosen[name] == true end
                return current == name
            end

            local function getValue()
                if multi then
                    local out = {}
                    for _, name in ipairs(options) do
                        if chosen[name] then table.insert(out, name) end
                    end
                    return out
                end
                return current
            end

            local function render()
                Dropdown.CurrentValue = getValue()
                if multi then
                    local v = Dropdown.CurrentValue
                    val.Text = (#v > 0) and table.concat(v, ", ") or "None"
                else
                    val.Text = current and tostring(current) or "None"
                end
            end

            local function fire(silent)
                render()
                if not silent and o.Callback then task.spawn(safe, o.Callback, Dropdown.CurrentValue) end
                changed()
            end

            render()

            local function build(list, close)
                local buttons = {}
                local gradients = {}

                local function restyle()
                    for name, b in pairs(buttons) do
                        local on = isOn(name)
                        b.BackgroundColor3 = on and WHITE or T.Item
                        gradients[name].Enabled = on
                    end
                end

                local box
                if #options > 8 then
                    box = new("TextBox", {
                        Size = UDim2.new(1, 0, 0, 24), LayoutOrder = 0, BackgroundColor3 = T.Item,
                        Text = "", PlaceholderText = "Search...", PlaceholderColor3 = T.Sub,
                        TextColor3 = T.Primary, Font = Enum.Font.Gotham, TextSize = 12,
                        ClearTextOnFocus = false, Parent = list,
                    }, { corner(7), pad(8, 8, 0, 0) })
                end

                if #options == 0 then
                    new("TextLabel", {
                        Size = UDim2.new(1, 0, 0, 24), BackgroundTransparency = 1, Text = "Empty",
                        TextColor3 = T.Sub, Font = Enum.Font.Gotham, TextSize = 12, Parent = list,
                    })
                end

                for i, name in ipairs(options) do
                    local b = new("TextButton", {
                        Size = UDim2.new(1, 0, 0, 26), LayoutOrder = i,
                        BackgroundColor3 = T.Item, Text = "", AutoButtonColor = false, Parent = list,
                    }, { corner(7) })
                    local g = gradient()
                    g.Enabled = false
                    g.Parent = b
                    local lbl = new("TextLabel", {
                        Size = UDim2.new(1, -12, 1, 0), Position = UDim2.new(0, 8, 0, 0),
                        BackgroundTransparency = 1, Text = tostring(name), TextColor3 = T.Text,
                        Font = Enum.Font.Gotham, TextSize = 12,
                        TextXAlignment = Enum.TextXAlignment.Left,
                        TextTruncate = Enum.TextTruncate.AtEnd, Parent = b,
                    })
                    buttons[name] = b
                    gradients[name] = g

                    pressFx(b, lbl)
                    b.MouseEnter:Connect(function()
                        if not isOn(name) then tween(b, 0.12, { BackgroundColor3 = T.ItemHi }) end
                        tween(lbl, 0.15, { Position = UDim2.new(0, 12, 0, 0) })
                    end)
                    b.MouseLeave:Connect(function()
                        if not isOn(name) then tween(b, 0.15, { BackgroundColor3 = T.Item }) end
                        tween(lbl, 0.2, { Position = UDim2.new(0, 8, 0, 0) })
                    end)
                    b.MouseButton1Click:Connect(function()
                        if multi then
                            chosen[name] = not chosen[name] or nil
                            restyle()
                            fire()
                        else
                            current = name
                            close()
                            fire()
                        end
                    end)
                end

                restyle()

                if box then
                    box:GetPropertyChangedSignal("Text"):Connect(function()
                        local q = string.lower(box.Text)
                        for name, b in pairs(buttons) do
                            b.Visible = (q == "") or (string.find(string.lower(tostring(name)), q, 1, true) ~= nil)
                        end
                    end)
                end
            end

            btn.MouseButton1Click:Connect(function()
                if panelOwner == Dropdown then
                    closePanel()
                    return
                end
                tween(arrow, 0.3, { Rotation = 180 }, EASE.Back, DIR.Out)
                openPanel(Dropdown, o.Name, build, function()
                    tween(arrow, 0.3, { Rotation = 0 }, EASE.Back, DIR.Out)
                end)
            end)

            function Dropdown:Set(v, silent)
                if multi then
                    chosen = {}
                    if type(v) == "string" then v = { v } end
                    for _, name in ipairs(v or {}) do chosen[name] = true end
                else
                    if type(v) == "table" then v = v[1] end
                    current = v
                end
                fire(silent)
            end
            function Dropdown:Get() return Dropdown.CurrentValue end

            function Dropdown:Refresh(newOptions)
                options = newOptions or {}
                if panelOwner == Dropdown then closePanel() end
                if multi then
                    for name in pairs(chosen) do
                        if not table.find(options, name) then chosen[name] = nil end
                    end
                else
                    if current ~= nil and not table.find(options, current) then current = nil end
                end
                render()
                changed()
            end

            register(o.Flag, Dropdown)
            return Dropdown
        end

        --// keybind
        function obj:CreateKeybind(o)
            local function toKey(v)
                if typeof(v) == "EnumItem" then return v end
                if type(v) == "string" then
                    local ok, k = pcall(function() return Enum.KeyCode[v] end)
                    if ok then return k end
                end
                return nil
            end
            local key = toKey(o.CurrentKeybind)

            local r = row(30)
            rowLabel(r, o.Name, 80)

            local kStroke = new("UIStroke", { Color = T.Primary, Thickness = 1, Transparency = 0.5 })
            local box = new("TextButton", {
                Size = UDim2.fromOffset(70, 22), Position = UDim2.new(1, -80, 0.5, -11),
                BackgroundColor3 = T.Bg, Text = "", AutoButtonColor = false, Parent = r,
            }, { corner(6), kStroke })
            local lbl = new("TextLabel", {
                Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1,
                Text = key and key.Name or "None", TextColor3 = T.Primary,
                Font = Enum.Font.GothamMedium, TextSize = 11, Parent = box,
            })
            pressFx(box, lbl)

            local Keybind = { Type = "Keybind", CurrentValue = key and key.Name or "None" }
            local listening = false

            box.MouseButton1Click:Connect(function()
                listening = true
                lbl.Text = "..."
                tween(kStroke, 0.15, { Transparency = 0 })
            end)

            local function setKey(k, silent)
                key = k
                Keybind.CurrentValue = key and key.Name or "None"
                lbl.Text = Keybind.CurrentValue
                changed()
            end

            track(UserInputService.InputBegan:Connect(function(i, gp)
                if listening then
                    if i.UserInputType == Enum.UserInputType.Keyboard then
                        listening = false
                        tween(kStroke, 0.2, { Transparency = 0.5 })
                        if i.KeyCode == Enum.KeyCode.Escape then
                            setKey(nil)
                        else
                            setKey(i.KeyCode)
                        end
                    end
                    return
                end
                if gp or not key then return end
                if i.KeyCode == key and o.Callback then
                    if o.HoldToInteract then
                        task.spawn(safe, o.Callback, true)
                    else
                        task.spawn(safe, o.Callback)
                    end
                end
            end))

            if o.HoldToInteract then
                track(UserInputService.InputEnded:Connect(function(i)
                    if key and i.KeyCode == key and o.Callback then
                        task.spawn(safe, o.Callback, false)
                    end
                end))
            end

            function Keybind:Set(v) setKey(toKey(v)) end
            function Keybind:Get() return key end

            register(o.Flag, Keybind)
            return Keybind
        end

            local function build(list)
                local svBox = new("Frame", {
                    Size = UDim2.new(1, 0, 0, 90), BackgroundColor3 = Color3.fromHSV(h, 1, 1),
                    BorderSizePixel = 0, LayoutOrder = 1, Parent = list,
                }, { corner(6) })
                new("Frame", {
                    Size = UDim2.fromScale(1, 1), BackgroundColor3 = WHITE,
                    BorderSizePixel = 0, Parent = svBox,
                }, {
                    corner(6),
                    new("UIGradient", { Transparency = NumberSequence.new(0, 1) }),
                })
                new("Frame", {
                    Size = UDim2.fromScale(1, 1), BackgroundColor3 = Color3.new(0, 0, 0),
                    BorderSizePixel = 0, Parent = svBox,
                }, {
                    corner(6),
                    new("UIGradient", { Rotation = 90, Transparency = NumberSequence.new(1, 0) }),
                })
                local svCursor = new("Frame", {
                    Size = UDim2.fromOffset(10, 10), AnchorPoint = Vector2.new(0.5, 0.5),
                    Position = UDim2.fromScale(s, 1 - v), BackgroundTransparency = 1,
                    ZIndex = 3, Parent = svBox,
                }, { corner(5), new("UIStroke", { Color = WHITE, Thickness = 1.5 }) })
                local svHit = new("TextButton", {
                    Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Text = "",
                    ZIndex = 4, Parent = svBox,
                })

                local hueBar = new("Frame", {
                    Size = UDim2.new(1, 0, 0, 14), BackgroundColor3 = WHITE,
                    BorderSizePixel = 0, LayoutOrder = 2, Parent = list,
                }, {
                    corner(6),
                    new("UIGradient", {
                        Color = ColorSequence.new({
                            ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 0, 0)),
                            ColorSequenceKeypoint.new(1 / 6, Color3.fromRGB(255, 255, 0)),
                            ColorSequenceKeypoint.new(2 / 6, Color3.fromRGB(0, 255, 0)),
                            ColorSequenceKeypoint.new(3 / 6, Color3.fromRGB(0, 255, 255)),
                            ColorSequenceKeypoint.new(4 / 6, Color3.fromRGB(0, 0, 255)),
                            ColorSequenceKeypoint.new(5 / 6, Color3.fromRGB(255, 0, 255)),
                            ColorSequenceKeypoint.new(1, Color3.fromRGB(255, 0, 0)),
                        }),
                    }),
                })
                local hueCursor = new("Frame", {
                    Size = UDim2.new(0, 4, 1, 4), AnchorPoint = Vector2.new(0.5, 0.5),
                    Position = UDim2.new(h, 0, 0.5, 0), BackgroundColor3 = WHITE,
                    BorderSizePixel = 0, ZIndex = 3, Parent = hueBar,
                }, { corner(2), new("UIStroke", { Color = Color3.new(0, 0, 0), Thickness = 1 }) })
                local hueHit = new("TextButton", {
                    Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Text = "",
                    ZIndex = 4, Parent = hueBar,
                })

                local hexBox = new("TextBox", {
                    Size = UDim2.new(1, 0, 0, 24), LayoutOrder = 3, BackgroundColor3 = T.Item,
                    Text = hex(), TextColor3 = T.Primary, Font = Enum.Font.GothamMedium,
                    TextSize = 12, ClearTextOnFocus = false, Parent = list,
                }, { corner(7) })

                local function refreshVisuals()
                    svBox.BackgroundColor3 = Color3.fromHSV(h, 1, 1)
                    svCursor.Position = UDim2.fromScale(s, 1 - v)
                    hueCursor.Position = UDim2.new(h, 0, 0.5, 0)
                    hexBox.Text = hex()
                end

                local dragSV, dragHue = false, false

                local function updSV(pos)
                    local ap, as = svBox.AbsolutePosition, svBox.AbsoluteSize
                    if as.X <= 0 or as.Y <= 0 then return end
                    s = math.clamp((pos.X - ap.X) / as.X, 0, 1)
                    v = 1 - math.clamp((pos.Y - ap.Y) / as.Y, 0, 1)
                    push()
                    refreshVisuals()
                end

                local function updHue(pos)
                    local ap, as = hueBar.AbsolutePosition, hueBar.AbsoluteSize
                    if as.X <= 0 then return end
                    h = math.clamp((pos.X - ap.X) / as.X, 0, 1)
                    push()
                    refreshVisuals()
                end

                local function isPress(i)
                    return i.UserInputType == Enum.UserInputType.MouseButton1
                        or i.UserInputType == Enum.UserInputType.Touch
                end

                svHit.InputBegan:Connect(function(i)
                    if isPress(i) then
                        dragSV = true
                        updSV(i.Position)
                    end
                end)
                hueHit.InputBegan:Connect(function(i)
                    if isPress(i) then
                        dragHue = true
                        updHue(i.Position)
                    end
                end)

                local c1 = UserInputService.InputChanged:Connect(function(i)
                    if i.UserInputType == Enum.UserInputType.MouseMovement
                    or i.UserInputType == Enum.UserInputType.Touch then
                        if dragSV then updSV(i.Position) end
                        if dragHue then updHue(i.Position) end
                    end
                end)
                local c2 = UserInputService.InputEnded:Connect(function(i)
                    if isPress(i) then
                        dragSV, dragHue = false, false
                    end
                end)

                hexBox.FocusLost:Connect(function()
                    local code = string.match(hexBox.Text, "^#?(%x%x%x%x%x%x)$")
                    if code then
                        local rr = tonumber(string.sub(code, 1, 2), 16)
                        local gg = tonumber(string.sub(code, 3, 4), 16)
                        local bb = tonumber(string.sub(code, 5, 6), 16)
                        h, s, v = Color3.fromRGB(rr, gg, bb):ToHSV()
                        push()
                    end
                    refreshVisuals()
                end)

                return function()
                    c1:Disconnect()
                    c2:Disconnect()
                end
            end

            swatch.MouseButton1Click:Connect(function()
                if panelOwner == Picker then
                    closePanel()
                    return
                end
                openPanel(Picker, o.Name, build)
            end)

            function Picker:Set(c, silent)
                if typeof(c) ~= "Color3" then return end
                h, s, v = c:ToHSV()
                push(silent)
                if panelOwner == Picker then closePanel() end
            end
            function Picker:Get() return Picker.CurrentValue end

            Picker._encode = function(c)
                return {
                    math.floor(c.R * 255 + 0.5),
                    math.floor(c.G * 255 + 0.5),
                    math.floor(c.B * 255 + 0.5),
                }
            end
            Picker._decode = function(t)
                return Color3.fromRGB(t[1] or 255, t[2] or 255, t[3] or 255)
            end

            register(o.Flag, Picker)
            return Picker
        end

        return nextOrder
    end

    --// tabs
    local tabs, active = {}, nil
    local switchId = 0
    local pillPlaced = false

    local function setTabStyle(t, on)
        tween(t.lbl, 0.25, { TextColor3 = on and WHITE or T.Sub })
        if on then
            tween(t.btn, 0.15, { BackgroundTransparency = 1 })
            local target = UDim2.fromOffset(8, 8 + (t.index - 1) * 34)
            if not pillPlaced then
                pillPlaced = true
                pill.Position = target
                pill.Visible = true
                tween(pill, 0.4, { BackgroundTransparency = 0 })
            else
                tween(pill, 0.38, { Position = target }, EASE.Back, DIR.Out)
            end
        end
    end

    local function selectTab(t)
        if active == t then return end
        switchId = switchId + 1
        local myId = switchId
        closePanel()

        local prev = active
        active = t
        if prev then setTabStyle(prev, false) end
        setTabStyle(t, true)

        -- old content out
        if prev then
            tween(contentWrap, 0.12, {
                GroupTransparency = 1, Position = contentBase + UDim2.fromOffset(-12, 0),
            }, EASE.Quad, DIR.In)
            task.wait(0.13)
            if myId ~= switchId or destroyed then return end
        end

        -- swap page, new content in
        for _, other in ipairs(tabs) do
            other.page.Visible = (other == t)
        end
        contentWrap.GroupTransparency = 1
        contentWrap.Position = contentBase + UDim2.fromOffset(16, 0)
        tween(contentWrap, 0.32, { GroupTransparency = 0, Position = contentBase }, EASE.Quint, DIR.Out)
    end

    function Window:CreateTab(name, icon)
        local idx = #tabs + 1
        local Tab = { Name = name }

        local btn = new("TextButton", {
            Size = UDim2.new(1, 0, 0, 28), LayoutOrder = idx,
            BackgroundColor3 = T.ItemHi, BackgroundTransparency = 1,
            Text = "", AutoButtonColor = false, Parent = sidebar,
        }, { corner(8) })

        local iconSize = 0
        if icon then
            iconSize = 18
            local img = icon
            if type(icon) == "number" then img = "rbxassetid://" .. tostring(icon) end
            new("ImageLabel", {
                Size = UDim2.fromOffset(14, 14), Position = UDim2.new(0, 8, 0.5, -7),
                BackgroundTransparency = 1, Image = tostring(img), Parent = btn,
            })
        end
        local lbl = new("TextLabel", {
            Size = UDim2.new(1, -iconSize, 1, 0), Position = UDim2.fromOffset(iconSize, 0),
            BackgroundTransparency = 1, Text = tostring(name), TextColor3 = T.Sub,
            Font = Enum.Font.GothamMedium, TextSize = 12, TextTruncate = Enum.TextTruncate.AtEnd,
            Parent = btn,
        })

        local page = new("ScrollingFrame", {
            Name = "Page_" .. tostring(name), Size = UDim2.fromScale(1, 1),
            BackgroundTransparency = 1, BorderSizePixel = 0, ScrollBarThickness = 3,
            ScrollBarImageColor3 = T.Primary, CanvasSize = UDim2.new(),
            AutomaticCanvasSize = Enum.AutomaticSize.Y, Visible = false, Parent = contentWrap,
        }, {
            pad(6, 6, 2, 4),
            new("UIListLayout", { Padding = UDim.new(0, 5), SortOrder = Enum.SortOrder.LayoutOrder }),
        })

        local rec = { name = name, index = idx, btn = btn, lbl = lbl, page = page }
        tabs[idx] = rec

        pressFx(btn, lbl)
        btn.MouseEnter:Connect(function()
            if active ~= rec then tween(btn, 0.15, { BackgroundTransparency = 0.5 }) end
        end)
        btn.MouseLeave:Connect(function()
            tween(btn, 0.2, { BackgroundTransparency = 1 })
        end)
        btn.MouseButton1Click:Connect(function() selectTab(rec) end)

        local nextOrder = attach(Tab, page)

        -- collapsible section, elements created from it go inside
        function Tab:CreateSection(sName, defaultOpen)
            local Section = { Name = sName }
            local open = defaultOpen and true or false

            local hStroke = new("UIStroke", { Color = T.Primary, Thickness = 1, Transparency = 0.7 })
            local header = new("TextButton", {
                Size = UDim2.new(1, 0, 0, 28), LayoutOrder = nextOrder(),
                BackgroundColor3 = T.Item, Text = "", AutoButtonColor = false, Parent = page,
            }, { corner(8), hStroke })

            local title = new("TextLabel", {
                Size = UDim2.new(1, -40, 1, 0), Position = UDim2.new(0, 10, 0, 0),
                BackgroundTransparency = 1, Text = tostring(sName), TextColor3 = WHITE,
                Font = Enum.Font.GothamBold, TextSize = 12,
                TextXAlignment = Enum.TextXAlignment.Left, Parent = header,
            }, { gradient() })

            local arrow = new("TextLabel", {
                Size = UDim2.fromOffset(20, 28), Position = UDim2.new(1, -26, 0, 0),
                BackgroundTransparency = 1, Text = ">", Rotation = open and 180 or 0,
                TextColor3 = T.Primary, Font = Enum.Font.GothamBold, TextSize = 14, Parent = header,
            })

            local glayout = new("UIListLayout", {
                Padding = UDim.new(0, 5), SortOrder = Enum.SortOrder.LayoutOrder,
            })
            local group = new("Frame", {
                Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y,
                LayoutOrder = nextOrder(), BackgroundTransparency = 1,
                Visible = open, Parent = page,
            }, { glayout })

            pressFx(header, title)
            header.MouseEnter:Connect(function()
                tween(header, 0.15, { BackgroundColor3 = T.ItemHi })
                tween(hStroke, 0.15, { Transparency = 0.2 })
            end)
            header.MouseLeave:Connect(function()
                tween(header, 0.2, { BackgroundColor3 = T.Item })
                tween(hStroke, 0.2, { Transparency = 0.7 })
            end)

            local function setOpen(v)
                open = v and true or false
                tween(arrow, 0.35, { Rotation = open and 180 or 0 }, EASE.Back, DIR.Out)
                animateGroup(group, glayout, open)
            end
            header.MouseButton1Click:Connect(function() setOpen(not open) end)

            function Section:SetOpen(v) setOpen(v) end
            function Section:Toggle() setOpen(not open) end
            function Section:IsOpen() return open end

            attach(Section, group)
            return Section
        end

        function Tab:Select() selectTab(rec) end

        -- first tab opens by default
        if idx == 1 then selectTab(rec) end

        return Tab
    end

    function Window:SelectTab(nameOrTab)
        local wanted = type(nameOrTab) == "table" and nameOrTab.Name or nameOrTab
        for _, t in ipairs(tabs) do
            if t.name == wanted then selectTab(t) return end
        end
    end

    --// open / close
    local opened, openToken = true, 0

    local function setOpen(v)
        opened = v
        openToken = openToken + 1
        local myTok = openToken
        if v then
            main.Visible = true
            tween(mainScale, 0.45, { Scale = 1 }, EASE.Back, DIR.Out)
        else
            closePanel()
            local tw = tween(mainScale, 0.2, { Scale = 0.8 }, EASE.Quad, DIR.In)
            tw.Completed:Connect(function()
                if myTok == openToken then main.Visible = false end
            end)
        end
    end

    function Window:SetVisible(v) setOpen(v and true or false) end
    function Window:Toggle() setOpen(not opened) end
    function Window:IsVisible() return opened end
    function Window:SetTitle(t) titleLabel.Text = tostring(t) end

    --// open button (1:1)
    local openScale = nil
    local openCfg = cfg.OpenButton
    if openCfg == nil then openCfg = true end
    if openCfg then
        local oc = type(openCfg) == "table" and openCfg or {}
        local openStroke = glowStroke(1.5, 0.2)
        local openBtn = new("ImageButton", {
            Name = "OpenButton", Size = oc.Size or UDim2.fromOffset(42, 42),
            AnchorPoint = Vector2.new(0.5, 0.5), Position = oc.Position or UDim2.new(0, 33, 0, 91),
            BackgroundColor3 = T.Bg, BorderSizePixel = 0,
            Image = cfg.Logo or "", ScaleType = Enum.ScaleType.Fit,
            AutoButtonColor = false, Parent = gui,
        }, {
            corner(10), openStroke,
            new("UIAspectRatioConstraint", { AspectRatio = 1 }),
        })
        if not (cfg.Logo and cfg.Logo ~= "") then
            new("TextLabel", {
                Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Text = initial(),
                TextColor3 = WHITE, Font = Enum.Font.GothamBlack, TextSize = 20, Parent = openBtn,
            }, { gradient(45) })
        end
        spin(openStroke, 4)

        openScale = scaleFx(openBtn, 1.1, 0.88)
        local openDragged = makeDraggable(openBtn, openBtn, 6)

        openBtn.MouseButton1Click:Connect(function()
            if openDragged() then return end
            setOpen(not opened)
        end)
    end

    -- "-" hides the gui, open it again with the open button or the toggle key
    btnMin.MouseButton1Click:Connect(function()
        setOpen(false)
    end)

    --// destroy
    function Window:Destroy()
        if destroyed then return end
        destroyed = true
        if activeNotify == myNotify then activeNotify = nil end
        for _, c in ipairs(conns) do
            pcall(function()
                if typeof(c) == "RBXScriptConnection" then c:Disconnect() else c:Cancel() end
            end)
        end
        table.clear(conns)
        for flag in pairs(flagList) do
            if ReigaX.Flags[flag] == flagList[flag] then ReigaX.Flags[flag] = nil end
        end
        gui:Destroy()
    end

    --// warning before close
    local confirm = cfg.ConfirmClose
    if confirm == nil then confirm = true end
    local cc = type(confirm) == "table" and confirm or {}

    local warnData = nil
    local closing = false

    local function closeWarning(cb)
        if not warnData then
            if cb then cb() end
            return
        end
        local w = warnData
        warnData = nil
        tween(w.scale, 0.2, { Scale = 0.85 }, EASE.Quad, DIR.In)
        tween(w.box, 0.2, { GroupTransparency = 1 }, EASE.Quad, DIR.In)
        tween(w.backdrop, 0.25, { BackgroundTransparency = 1 }, EASE.Quad, DIR.In)
        task.delay(0.27, function()
            pcall(function() w.backdrop:Destroy() end)
            if cb then cb() end
        end)
    end

    local function doClose()
        if closing then return end
        closing = true
        if cfg.OnClose then pcall(cfg.OnClose) end
        closeWarning(function()
            tween(mainScale, 0.28, { Scale = 0.6 }, EASE.Back, DIR.In)
            if openScale then tween(openScale, 0.28, { Scale = 0 }, EASE.Back, DIR.In) end
            task.delay(0.3, function() Window:Destroy() end)
        end)
    end

    local function showWarning()
        if warnData or closing then return end
        closePanel()

        local backdrop = new("Frame", {
            Name = "Warning", Size = UDim2.fromScale(1, 1),
            BackgroundColor3 = Color3.new(0, 0, 0), BackgroundTransparency = 1,
            BorderSizePixel = 0, Active = true, ZIndex = 100, Parent = gui,
        })
        local box = new("CanvasGroup", {
            Size = UDim2.fromOffset(300, 150), AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.fromScale(0.5, 0.5), GroupTransparency = 1,
            BackgroundColor3 = T.Bg, BorderSizePixel = 0, Parent = backdrop,
        }, { corner(12), glowStroke(1.5, 0.2) })
        local boxScale = new("UIScale", { Scale = 0.8, Parent = box })

        warnData = { backdrop = backdrop, box = box, scale = boxScale }

        new("TextLabel", {
            Size = UDim2.new(1, 0, 0, 26), Position = UDim2.new(0, 0, 0, 10),
            BackgroundTransparency = 1, Text = cc.Title or "Warning!", TextColor3 = T.Danger,
            Font = Enum.Font.GothamBlack, TextSize = 18, Parent = box,
        })
        new("TextLabel", {
            Size = UDim2.new(1, -28, 0, 32), Position = UDim2.new(0, 14, 0, 40),
            BackgroundTransparency = 1,
            Text = cc.Text or "The script will be disabled and the frame will be deleted!",
            TextColor3 = T.Text, Font = Enum.Font.GothamMedium, TextSize = 12,
            TextWrapped = true, Parent = box,
        })
        new("TextLabel", {
            Size = UDim2.new(1, -28, 0, 18), Position = UDim2.new(0, 14, 0, 76),
            BackgroundTransparency = 1,
            Text = cc.Question or "Are you sure you want to delete the script?",
            TextColor3 = T.Sub, Font = Enum.Font.Gotham, TextSize = 11,
            TextWrapped = true, Parent = box,
        })

        local function dlgBtn(text, x, color, strokeColor)
            local b = new("TextButton", {
                Size = UDim2.fromOffset(128, 30), AnchorPoint = Vector2.new(0.5, 0.5),
                Position = UDim2.new(0, x + 64, 0, 123),
                BackgroundColor3 = color, Text = text, TextColor3 = T.Text,
                Font = Enum.Font.GothamBold, TextSize = 13, AutoButtonColor = true, Parent = box,
            }, { corner(8), new("UIStroke", { Color = strokeColor, Thickness = 1, Transparency = 0.3 }) })
            scaleFx(b, 1.06, 0.92)
            return b
        end

        local btnNo  = dlgBtn(cc.No or "No", 16, T.Item, T.Secondary)
        local btnYes = dlgBtn(cc.Yes or "Yes", 156, Color3.fromRGB(190, 55, 75), T.Danger)

        btnNo.MouseButton1Click:Connect(function() closeWarning() end)
        btnYes.MouseButton1Click:Connect(doClose)

        tween(backdrop, 0.25, { BackgroundTransparency = 0.45 })
        tween(box, 0.25, { GroupTransparency = 0 })
        tween(boxScale, 0.4, { Scale = 1 }, EASE.Back, DIR.Out)
    end

    -- close from the X button. skipConfirm = true closes right away
    function Window:Close(skipConfirm)
        if confirm and not skipConfirm then showWarning() else doClose() end
    end

    btnClose.MouseButton1Click:Connect(function() Window:Close() end)

    --// toggle key
    local toggleKey = cfg.ToggleKey
    if type(toggleKey) == "string" then
        local ok, k = pcall(function() return Enum.KeyCode[cfg.ToggleKey] end)
        toggleKey = ok and k or nil
    end
    if toggleKey then
        track(UserInputService.InputBegan:Connect(function(i, gp)
            if gp then return end
            if i.KeyCode == toggleKey then setOpen(not opened) end
        end))
    end

    --// intro
    if openScale then
        openScale.Scale = 0
        tween(openScale, 0.55, { Scale = 1 }, EASE.Back, DIR.Out)
    end
    setOpen(true)

    return Window
end

return ReigaX
